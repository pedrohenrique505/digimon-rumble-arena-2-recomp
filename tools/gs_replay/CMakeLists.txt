cmake_minimum_required(VERSION 3.21)
project(dra2_gs_replay LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(RUNTIME "${CMAKE_CURRENT_SOURCE_DIR}/../../external/PS2Recomp/ps2xRuntime" CACHE PATH "PS2Recomp runtime")
set(PARALLEL_GS_SOURCE "" CACHE PATH "Pinned paraLLEl-GS checkout")
add_library(cpu-gs STATIC "${RUNTIME}/src/lib/gs/gs_frontend.cpp" "${RUNTIME}/src/lib/gs/gs_cpu_backend.cpp" "${RUNTIME}/src/lib/gs/ps2_gs_memory.cpp")
target_include_directories(cpu-gs PUBLIC "${RUNTIME}/include" "${RUNTIME}/src/lib")
target_compile_definitions(cpu-gs PUBLIC PS2_RUNTIME_LOGS=0 AGRESSIVE_LOGS=0)
target_compile_options(cpu-gs PRIVATE -O2 -march=x86-64-v3)
add_executable(dra2-gs-replay replay.cpp)
target_link_libraries(dra2-gs-replay PRIVATE cpu-gs)
add_executable(dra2-gs-test-depth test_depth.cpp)
target_link_libraries(dra2-gs-test-depth PRIVATE cpu-gs)
add_executable(dra2-gs-test-constant-triangle test_constant_triangle.cpp)
target_link_libraries(dra2-gs-test-constant-triangle PRIVATE cpu-gs)
if(PARALLEL_GS_SOURCE)
 set(PARALLEL_GS_STANDALONE ON CACHE BOOL "" FORCE)
 set(GRANITE_AUDIO OFF CACHE BOOL "" FORCE)
 add_subdirectory("${PARALLEL_GS_SOURCE}" parallel-gs EXCLUDE_FROM_ALL)
 add_executable(dra2-gs-replay-vulkan replay.cpp)
 target_compile_definitions(dra2-gs-replay-vulkan PRIVATE DRA2_PARALLEL_GS=1)
 target_link_libraries(dra2-gs-replay-vulkan PRIVATE cpu-gs parallel-gs)
endif()
